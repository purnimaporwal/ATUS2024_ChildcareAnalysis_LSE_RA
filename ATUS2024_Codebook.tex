\begin{tcolorbox}[notebox]
\textbf{Note on variable documentation.}
Variable labels were extracted from the ATUS 2024 Stata \texttt{.do} files
(\texttt{atusresp\_2024.do}, \texttt{atusact\_2024.do}, etc.) and verified
against the official BLS data dictionary (\texttt{atus\_int\_codebk24.pdf}).
Reading Stata \texttt{.do} files in R is the equivalent of Stata's variable
labelling system: every variable name, what it measures, and what its values
mean is documented there. I used Claude (Anthropic) to assist with extracting
and organising the variable labels from these source files.
\end{tcolorbox}

\vspace{6pt}
\textbf{Five ATUS files used in this analysis:}
\begin{itemize}[nosep, leftmargin=1.4em]
  \item \texttt{atusresp\_2024.dat} --- Respondent file (175 variables, 7{,}669 obs.)
  \item \texttt{atusrost\_2024.dat} --- Roster file (8 variables)
  \item \texttt{atusact\_2024.dat} --- Activity file (30 variables)
  \item \texttt{atuswho\_2024.dat} --- Who file (5 variables)
  \item \texttt{atussum\_2024.dat} --- Activity Summary file (396 variables, 7{,}669 obs.)
\end{itemize}

\smallskip
\textbf{Primary merge key:} \texttt{TUCASEID} (14-digit case ID, present in all five files).\\
\textbf{Activity-level merge key:} \texttt{TUACTIVITY\_N} (links \texttt{act} $\leftrightarrow$ \texttt{who}).\\
\textbf{Person-level merge key:} \texttt{TULINENO} (links \texttt{who} $\leftrightarrow$ \texttt{rost}).

\section{Respondent File (\texttt{atusresp\_2024.dat}) --- 175 Variables}
%%%════════════════════════════════════════════════════════════════════════════

\begin{tcolorbox}[notebox]
One row per respondent (7{,}669 rows). Contains demographic, labour force,
earnings, family structure, and pre-computed time-use summary variables.

\subsection{Secondary Childcare (BLS Pre-computed)}

\begin{tcolorbox}[notebox]
\small Secondary childcare = respondent was doing something else as the primary
activity but was simultaneously responsible for a child.
\textbf{Important distinction:} \texttt{TRTCC} (two C's) = BLS secondary
childcare for HH and own non-HH children under 13 --- this is the childcare
variable used as M4 in this analysis.
\texttt{TRTCCC} (three C's) = time with customers, clients, and co-workers;
it is \emph{not} a childcare measure. Verify labels against the
\texttt{.do} file before use.

\section{Roster File (\texttt{atusrost\_2024.dat}) --- 8 Variables}
%%%════════════════════════════════════════════════════════════════════════════

\begin{tcolorbox}[notebox]
One row per household member per respondent. Used to identify the age of each
person present during activities (linked via \texttt{TUCASEID} +
\texttt{TULINENO} to the Who file). Key for the M2 child-present measure:
check whether any roster person with \texttt{TEAGE} $< 18$ appears in the Who
file for a given activity.
\end{tcolorbox}

\begin{longtable}{L{3cm} L{5.8cm} L{5.7cm}}
\toprule
\textbf{Variable} & \textbf{Description} & \textbf{Values} \\
\midrule \endhead \bottomrule \endfoot
\texttt{TUCASEID}  & ATUS Case ID & 14-digit string. Merge key \\
\rowcolor{rowgray} \texttt{TULINENO}  & Person line number & Links to Who file \\
\texttt{TEAGE}     & Age (edited) & In years \\
\rowcolor{rowgray} \texttt{TERRP}     & Relationship to respondent (edited) &
  18/19=Self; 20=Spouse; 21=Unmarried partner; 22=Own HH child;
  23=Grandchild; 24=Parent; 25=Brother/sister; 26=Other relative;
  27=Foster child; 28=Housemate; 29=Roomer/boarder; 30=Other nonrelative;
